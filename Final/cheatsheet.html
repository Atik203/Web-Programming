  // STAGE 0: State Variables OUTSIDE Function!
  let runningTotal = 0, sessionCount = 0;
  let itemHistory = [];

  function processApp() {
    // STAGE 1: Extract & Parse
    let name = document.getElementById("fName").value;
    let val  = parseFloat(document.getElementById("fVal").value);
    let out  = document.getElementById("msgOut");

    // STAGE 2: Mutate State & History
    runningTotal += val; sessionCount++;
    itemHistory.push(name);

    // STAGE 3: Compute Statistical Averages
    let avg = runningTotal / sessionCount;

    // STAGE 4: Evaluation Ladder (Highest First!)
    let feedback = "";
    if (runningTotal >= 100)      feedback = "Goal Achieved!";
    else if (runningTotal >= 60) feedback = "Good Progress!";
    else                         feedback = "Keep Working!";

    // Guard: Alert on excessive attempts
    if (sessionCount > 7 && runningTotal < 100) {
      feedback += " (High Attempts Warning!)";
    }

    // STAGE 5: Render DOM Feedback
    out.innerHTML = `Total: ${runningTotal} | Avg: ${avg.toFixed(2)} | ` +
      `History: [${itemHistory.join(", ")}] | Status: <b>${feedback}</b>`;

    // STAGE 6: Termination Lock
    if (runningTotal >= 100 || sessionCount >= 10) {
      document.getElementById("fVal").disabled = true;
      document.getElementById("btn").disabled = true;
    }
  }
</script>

Pattern 2 — Group Aggregations with HAVING:
-- HAVING filters aggregated groups (WHERE cannot!)
SELECT category, COUNT(*) AS total_items, AVG(rating) AS avg_rate
FROM records
GROUP BY category
HAVING COUNT(*) > 1 AND AVG(rating) > 4.4
ORDER BY avg_rate DESC;
Golden Rule: Every non-aggregated selected column MUST be in GROUP BY! Use HAVING for aggregate conditions, never WHERE!
